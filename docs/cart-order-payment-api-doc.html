Cart

interface IAddCartItemPayload {
  productId: string; // UUID
  optionIds?: string[]; // UUID[]
  variantOptionId?: string; // UUID
  variantOptionIds?: string[]; // UUID[]
  quantity?: number; // integer, 1..100
}

Business rule:
- At least one must exist: variantOptionId OR variantOptionIds OR optionIds
- Duplicate IDs in optionIds / variantOptionIds are not allowed

interface IUpdateCartItemPayload {
  quantity: number; // integer, 1..100
}

Order

type TPaymentMethod =
  | "CASH_ON_DELIVERY"
  | "BKASH"
  | "NAGAD"
  | "ROCKET"
  | "CREDIT_CARD"
  | "BANK_TRANSFER";

type TOrderStatus =
  | "PENDING"
  | "PROCESSING"
  | "CONFIRMED"
  | "SHIPPED"
  | "DELIVERED"
  | "CANCELLED"
  | "RETURNED"
  | "REFUNDED";

interface ICreateOrderPayload {
  shippingAddressId: string; // UUID
  paymentMethod: TPaymentMethod;
  notes?: string; // 1..2000 chars
  billingAddress?: Record<string, unknown>;
  shippingCost?: number; // >= 0
  tax?: number; // >= 0
}

interface IUpdateOrderStatusPayload {
  status: TOrderStatus;
  note?: string; // 1..500 chars
}

Payment

// Used by payment initiation routes:
interface IInitiatePaymentBody {}
// Strict empty object. Do not send extra keys.

// Used by callback/validation routes:
interface ISslCallbackPayload {
  tran_id?: string;
  transactionId?: string;
  val_id?: string;
  status?: string;
  amount?: string | number;
}

Business rule:
- tran_id OR transactionId is required
- If status means success (SUCCESS / VALID / VALIDATED), val_id is required

Review

interface ICreateReviewPayload {
  productId: string; // UUID
  rating: number; // integer, 1..5
  title?: string; // 1..255 chars
  comment?: string; // 1..2000 chars
}

interface IUpdateMyReviewPayload {
  rating?: number; // integer, 1..5
  title?: string; // 1..255 chars
  comment?: string; // 1..2000 chars
}

interface IModerateReviewPayload {
  isApproved?: boolean;
  isVerifiedPurchase?: boolean;
}

interface IReplyReviewPayload {
  adminReply: string | null; // null = remove reply
}

Business rule:
- update/moderate body cannot be empty
- for adminReply, blank string becomes null

User

type TRole = "ADMIN" | "USER" | "MANAGER";

type TUserStatus = "ACTIVE" | "BLOCKED" | "DELETED";

interface IAdminUpdateUserPayload {
  name?: string; // 2..50 chars
  role?: TRole;
  emailVerified?: boolean;
  status?: TUserStatus;
  phone?: string; // "+8801XXXXXXXXX" or "01XXXXXXXXX"

interface IUpdateMePayload {
  name?: string; // 2..50 chars
  phone?: string; // "+8801XXXXXXXXX" or "01XXXXXXXXX"

  status?: "DELETED"; // current validation only accepts DELETED
}

interface ICreateAddressPayload {
  label?: string; // 2..30 chars
  recipient?: string; // 2..100 chars
  phone?: string; // "+8801XXXXXXXXX" or "01XXXXXXXXX"
  street: string; // 5..255 chars
  city: string; // 2..100 chars
  state?: string; // max 100 chars
  zipCode?: string; // max 20 chars
  country?: string; // max 100 chars
  isDefault?: boolean;
}

interface IUpdateAddressPayload {
  label?: string; // 2..30 chars
  recipient?: string; // 2..100 chars
  phone?: string; // "+8801XXXXXXXXX" or "01XXXXXXXXX"
  street?: string; // 5..255 chars
  city?: string; // 2..100 chars
  state?: string; // max 100 chars
  zipCode?: string; // max 20 chars
  country?: string; // max 100 chars
  isDefault?: boolean;

Business rule:
- update address body cannot be empty

Quick Frontend Notes

For protected endpoints, include JWT in Authorization header. Public SSL callback routes should be called by payment gateway server.

Most ids are UUID values in this project schema (orderId, itemId, paymentId, productId, reviewId, addressId, userId).

For payment initiation routes, send an empty JSON object: {}.
